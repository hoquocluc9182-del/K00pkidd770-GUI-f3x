-- Bộ hàm Core tương tác hệ thống chuẩn F3X (Có SetLocked)
local function resize(part, size, cf)
    remote:InvokeServer("SyncResize", {{["Part"] = part, ["CFrame"] = cf, ["Size"] = size}})
end

local function color(part, color)
    remote:InvokeServer("SyncColor", {{["Part"] = part, ["Color"] = color, ["UnionColoring"] = false}})
end

local function setanchor(part, boolean)
    remote:InvokeServer("SyncAnchor", {{["Part"] = part, ["Anchored"] = boolean}})
end

local function lock(part, boolean)
    remote:InvokeServer("SetLocked", {part}, boolean)
end

local function name(part, stringa)
    remote:InvokeServer("SetName", {part}, stringa)
end

local function syncmaterial(part, mate)
    remote:InvokeServer("SyncMaterial", {{["Part"] = part, ["Material"] = mate}})
end

local function transparency(part, trans)
    remote:InvokeServer("SyncMaterial", {{["Part"] = part, ["Transparency"] = trans}})
end

local function setcollision(part, booleana)
    remote:InvokeServer("SyncCollision", {{["Part"] = part, ["CanCollide"] = booleana}})
end

local function createdecal(part, side)
    remote:InvokeServer("CreateTextures", {{["Part"] = part, ["Face"] = side, ["TextureType"] = "Decal"}})
end

local function setdecal(part, asset, side)
    remote:InvokeServer("SyncTexture", {{["Part"] = part, ["Face"] = side, ["TextureType"] = "Decal", ["Texture"] = "rbxassetid://".. asset}})
end

local function makemesh(part)
    remote:InvokeServer("CreateMeshes", {{["Part"] = part}})
end

local function syncmeshid(part, id)
    remote:InvokeServer("SyncMesh", {{["Part"] = part, ["MeshId"] = "rbxassetid://"..id}})
end

local function syncmeshtexture(part, id)
    remote:InvokeServer("SyncMesh", {{["Part"] = part, ["TextureId"] = "rbxassetid://"..id}})
end

local function syncmeshsize(part, vectora)
    remote:InvokeServer("SyncMesh", {{["Part"] = part, ["Scale"] = vectora}})
end

-- Hàm tạo hiệu ứng khói bảo vệ che mắt
local function AddSmokeToPart(part)
    remote:InvokeServer("CreateEffects", {{["Part"] = part, ["EffectType"] = "Smoke"}})
    remote:InvokeServer("SyncSmoke", {{
        ["Part"] = part, 
        ["Color"] = Color3.fromRGB(220, 220, 220), 
        ["Opacity"] = 0.8, 
        ["RiseVelocity"] = 15, 
        ["Size"] = 25
    }})
end

local function delete(part)
    remote:InvokeServer("Remove", {part})
end

-- Hàm tạo Part gốc
local function SpawnPart(partType, partName, size, targetCF, partColor)
    local part = remote:InvokeServer("CreatePart", partType or "Normal", targetCF, workspace)
    task.wait(0.05)
    if part then
        name(part, partName)
        resize(part, size, targetCF)
        color(part, partColor)
        setanchor(part, true)
        lock(part, true) -- Khóa đóng băng ngay lập tức
    end
    return part
end
