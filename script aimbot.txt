local function CreateToggle(name, default, callback)
    local Button = Instance.new("TextButton")
    Button.Name = name
    Button.Parent = ContentFrame
    Button.Size = UDim2.new(1, 0, 0, 38)
    Button.BackgroundColor3 = default and Color3.fromRGB(75, 75, 75) or Color3.fromRGB(45, 45, 45)
    Button.Font = Enum.Font.Arcade
    Button.Text = name .. ": " .. (default and "ON" or "OFF")
    Button.TextColor3 = Color3.fromRGB(255, 255, 255)
    Button.TextSize = 16
    Button.BorderSizePixel = 0
    
    local enabled = default
    Button.MouseButton1Click:Connect(function()
        enabled = not enabled
        Button.BackgroundColor3 = enabled and Color3.fromRGB(75, 75, 75) or Color3.fromRGB(45, 45, 45)
        Button.Text = name .. ": " .. (enabled and "ON" or "OFF")
        callback(enabled)
    end)
end

local isOpened = true

    if isOpened then
        isOpened = false
        local CloseTween = TweenService:Create(MainFrame, CloseTweenInf, {Size = UDim2.new(0, 0, 0, 0)})
        CloseTween:Play()
        CloseTween.Completed:Connect(function()
            if not isOpened then MainFrame.Visible = false end
        end)
    else
        isOpened = true
        MainFrame.Visible = true
        local OpenTween = TweenService:Create(MainFrame, TweenInf, {Size = OriginalSize})
        OpenTween:Play()
    end
end)

-- 4. ĐẦY ĐỦ LOGIC TÍNH NĂNG GỐC - ĐÃ TỐI ƯU FPS
-- ==========================================
local Camera = game:GetService("Workspace").CurrentCamera
local Players = game:GetService("Players")
local LocalPlayer = Players.LocalPlayer

local AimbotEnabled = false
local FOVVisible = false
local FOVRadius = 120

local ESPBoxes = false
local ESPTracers = false

local FOVCircle = Drawing.new("Circle")
FOVCircle.Thickness = 1.5
FOVCircle.NumSides = 32 
FOVCircle.Radius = FOVRadius
FOVCircle.Filled = false
FOVCircle.Visible = false
FOVCircle.Color = VisualColor

CreateToggle("AIMBOT", false, function(v) AimbotEnabled = v end)
CreateToggle("SHOW FOV", false, function(v) FOVVisible = v; FOVCircle.Visible = v end)
CreateToggle("ESP BOXES", false, function(v) ESPBoxes = v end)
CreateToggle("ESP TRACERS", false, function(v) ESPTracers = v end)

local ActiveESP = {}

local function ClearESP(player)
    if ActiveESP[player] then
        ActiveESP[player].Box.Visible = false
        ActiveESP[player].Box:Remove()
        ActiveESP[player].Tracer.Visible = false
        ActiveESP[player].Tracer:Remove()
        ActiveESP[player] = nil
    end
end

local function ApplyESP(player)
    if player == LocalPlayer then return end
    ClearESP(player)
    
    ActiveESP[player] = {
        Box = Drawing.new("Square"),
        Tracer = Drawing.new("Line")
    }
    ActiveESP[player].Box.Thickness = 1.5
    ActiveESP[player].Box.Filled = false
    ActiveESP[player].Box.Visible = false
    ActiveESP[player].Box.Color = VisualColor
    
    ActiveESP[player].Tracer.Thickness = 1.5
    ActiveESP[player].Tracer.Visible = false
    ActiveESP[player].Tracer.Color = VisualColor
end

for _, p in pairs(Players:GetPlayers()) do ApplyESP(p) end
Players.PlayerAdded:Connect(ApplyESP)
Players.PlayerRemoving:Connect(ClearESP)

local LastFpsTime = 0 
local FrameCount = 0

local function IsVisible(targetCharacter)
    if not targetCharacter or not targetCharacter:FindFirstChild("Head") then return false end
    local parts = Camera:GetPartsObscuringTarget({Camera.CFrame.Position, targetCharacter.Head.Position}, {LocalPlayer.Character, targetCharacter})
    return #parts == 0
end

local function GetClosestPlayer()
    local ClosestTarget = nil
    local MaxDistance = FOVRadius
    local MousePosition = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2)
    
    for _, player in pairs(Players:GetPlayers()) do
        if player ~= LocalPlayer and player.Character then
            local char = player.Character
            local root = char:FindFirstChild("HumanoidRootPart")
            local hum = char:FindFirstChild("Humanoid")
            
            if root and hum and hum.Health > 0 then
                local ScreenPosition, OnScreen = Camera:WorldToViewportPoint(root.Position)
                if OnScreen then
                    local Distance = (Vector2.new(ScreenPosition.X, ScreenPosition.Y) - MousePosition).Magnitude
                    if Distance < MaxDistance and IsVisible(char) then
                        ClosestTarget = player
                        MaxDistance = Distance
                    end
                end
            end
        end
    end
    return ClosestTarget
end

-- VÒNG LẶP CHÍNH MƯỢT MÀ, GIẢM TẢI CHO MÁY
RunService.RenderStepped:Connect(function(deltaTime)
    -- Thuật toán tính FPS trung bình mượt hơn, tránh nhảy số ảo liên tục
    FrameCount = FrameCount + 1
    LastFpsTime = LastFpsTime + deltaTime
    if LastFpsTime >= 0.5 then 
        local ActualFPS = math.floor(FrameCount / LastFpsTime)
        FpsLabel.Text = "FPS: " .. tostring(ActualFPS)
        FrameCount = 0
        LastFpsTime = 0
    end

    if FOVVisible then
        FOVCircle.Position = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2)
    end

    if AimbotEnabled then
        local Target = GetClosestPlayer()
        if Target and Target.Character and Target.Character:FindFirstChild("Head") then
            Camera.CFrame = CFrame.new(Camera.CFrame.Position, Target.Character.Head.Position)
        end
    end

    -- Quét hệ thống ESP có giới hạn khoảng cách để tránh drop FPS xuống 5
    for player, esp in pairs(ActiveESP) do
        if (ESPBoxes or ESPTracers) and player.Character and LocalPlayer.Character then
            local root = player.Character:FindFirstChild("HumanoidRootPart")
            local hum = player.Character:FindFirstChild("Humanoid")
            local localRoot = LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
            
            -- Chỉ vẽ ESP nếu mục tiêu còn sống và không ở quá xa (Tránh lag máy)
            if root and hum and hum.Health > 0 and localRoot and (root.Position - localRoot.Position).Magnitude < 800 then
                local Position, OnScreen = Camera:WorldToViewportPoint(root.Position)
                
                if OnScreen then
                    if ESPBoxes then
                        local SizeX = Camera.ViewportSize.X / Position.Z * 2.5
                        local SizeY = Camera.ViewportSize.Y / Position.Z * 2.5
                        esp.Box.Size = Vector2.new(SizeX, SizeY)
                        esp.Box.Position = Vector2.new(Position.X - SizeX / 2, Position.Y - SizeY / 2)
                        esp.Box.Visible = true
                    else
                        esp.Box.Visible = false
                    end
                    
                    if ESPTracers then
                        esp.Tracer.From = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y)
                        esp.Tracer.To = Vector2.new(Position.X, Position.Y)
                        esp.Tracer.Visible = true
                    else
                        esp.Tracer.Visible = false
                    end
                else
                    esp.Box.Visible = false
                    esp.Tracer.Visible = false
                end
            else
                esp.Box.Visible = false
                esp.Tracer.Visible = false
            end
        else
            esp.Box.Visible = false
            esp.Tracer.Visible = false
        end
    end
end)
